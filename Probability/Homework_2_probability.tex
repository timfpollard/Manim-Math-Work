\section*{(1)}
A machine produces photo detectors in pairs. Tests show that the first photo
detector is acceptable with probability 3/5. When the first photo detector is acceptable,
the second photo detector is acceptable with probability 4/5. Otherwise, if the first photo
detector is defective, the second photo detector is acceptable with probability 2/5.

\begin{enumerate}
\item[(a)] What is the probability that exactly one photo detector of a pair is acceptable?
\item[] \textcolor{blue}{A = detector 1, B = detector 2.}
\item[] \textcolor{blue}{$P($singular detector$) = P(A \cap B^\mathsf{c}) + P(A^\mathsf{c} \cap B)$}
\item[] \textcolor{blue}{$P(A \cap B^\mathsf{c}) = P(B^\mathsf{c}|A)\times P(A) = \frac{1}{5} \times \frac{3}{5}= \frac{3}{25}= 0.12$}
\item[] \textcolor{blue}{$P(A^\mathsf{c} \cap B) = P(B|A^\mathsf{c})\times P(A^\mathsf{c}) = \frac{2}{5}\times \frac{2}{5} = \frac{4}{25}= 0.16$}
\item[] \textcolor{blue}{$P($singular detector$) = 0.12+0.16=0.28$}
\item[(b)] What is the probability that both photo detectors in a pair are defective?
\item[] \textcolor{blue}{$P(A^\mathsf{c} \cap B^\mathsf{c}) = P(B^\mathsf{c}|A^\mathsf{c})\times P(A^\mathsf{c}) = \frac{3}{5} \times \frac{2}{5} = \frac{6}{25} = 0.24$}
\end{enumerate}

\section*{(2)}
 Box 1 contains 1 white and 999 red balls. Box 2 contains 1 red and 999 white
balls. A ball is picked from a randomly selected box. If the ball is red, what is the
probability that it came from box 1? 

\textcolor{blue}{\(P(\text{Box}_1| \text{Red})= \frac{P(\text{Box}_1 \cap \text{Red})}{P(\text{Red})} = \frac{\frac{999}{2000}}{\frac{1000}{2000}} = 0.999\)}
\section*{(3)}
 Box 1 contains 1000 bulbs of which 10 percent are defective. Box 2 contains 2000
bulbs of which 5 percent are defective. Two bulbs are picked from a randomly selected
box
\begin{enumerate}
    \item[(a)] Find the probability that both bulbs are defective.
    \item[] \textcolor{blue}{$P($Both defective$) = \frac{1}{2}(\frac{100}{1000}\times\frac{99}{999})+\frac{1}{2}(\frac{100}{2000}\times \frac{99}{1999}) \approx 0.00619$}
    \item[(b)] Assuming that both are defective, find the probability that they came from Box 1.
    \item[] \textcolor{blue}{$P(\text{Both Box}_1\,|\,\text{Both defective}) = \frac{P(\text{Both Box}_1 \cap \text{Both defective})}{P(\text{Both defective})} = \frac{\frac{1}{2}(\frac{100}{1000}\times\frac{99}{999})}{0.0560} \approx 0.800$}
\end{enumerate}

\section*{(5)}
There is a collection of field goal kickers, which can be divided into two groups 1
and 2. Group i has 3i shooters. On any kick, a kicker from group i will kick a field goal
with probability 1/(i + 1), independent of the outcome of any other kicks by that kicker
or any other kicker.
\begin{enumerate}
\item[(a)] A kicker is selected at random from among all the kickers and attempts one field goal. Let $K$ be the event that a field goal is kicked. Find $P(K)$.
\item[] \textcolor{blue}{$P(K) = P(\text{group 1} \cap K) + P(\text{group 2} \cap K) = (\frac{3}{9}\times \frac{1}{1+1}) + (\frac{6}{9}\times \frac{1}{2+1})= \frac{7}{18}= 0.38\bar{8}$}
\item[(b)] Two kickers are selected at random. For $j = 1, 2$, let $K_j$ be the event that kicker $j$ kicks a field goal. Find $P(K_1 \cap K_2)$. Are $K_1$ and $K_2$ independent events?
\item[] \textcolor{blue}{\small $P(K_1 \cap K_2) = P(K_1 \cap K_2 | G_1G_1) P(G_1G_1) + P(K_1 \cap K_2 | G_2G_2) P(G_2G_2) + P(K_1 \cap K_2 |G_1 G_2) P(G_1  G_2) $
\item[] 
$\left(\frac{1}{2}\right)^{2}\left(\frac{3}{9}\times\frac{2}{8}\right) + \left(\frac{1}{3}\right)^{2}\left(\frac{6}{9}\times\frac{5}{8}\right) + \left(\frac{1}{2}\times\frac{1}{3}\right)\left(2\times\frac{6}{9}\times\frac{3}{8}\right) = \frac{65}{432} \approx 0.150$ }

\textcolor{blue}{$\left(\frac{7}{18}\right)^2 = \frac{49}{324} \neq \frac{65}{432}, \therefore K_1 \text{ and } K_2 \text{ are not independent}$}

\item[(c)] A kicker is selected at random and attempts 10 field goals. Let $M$ be the number of misses. Find $P(M = 5)$.
\item[] 
\textcolor{blue}{$P(M=5)= P(M=5|G_1)P(G_1) +P(M=5|G_2)P(G_2)$}
\\\\
\textcolor{blue}{$P(M=5|G_1) = \binom{10}{5}\times (\frac{1}{2})^5 (1-\frac{1}{2})^5 = \frac{63}{256} \approx 0.246$}

\textcolor{blue}{$P(M=5|G_2) = \binom{10}{5}\times (\frac{1}{3})^5 (1-\frac{1}{3})^5 = \frac{8064}{59049} \approx 0.137$}

\textcolor{blue}{$P(M=5)= (\frac{63}{256})(\frac{1}{2}) +(\frac{8064}{59049} )(\frac{1}{3}) \approx 0.191$}
\end{enumerate}
