%------------ Bayes Rule & Axioms of Conditional Probability ---------------
\begin{tikzpicture}
\node [mybox] (box){%
    \begin{minipage}{0.3\textwidth}
    Bayes Rule: $P(B|A)= \frac{P(A|B)P(B)}{P(A)}$\\
    Conditional Probability: $P(A|B) = \frac{P(A \cap B)}{P(B)}$ \\
    $P(A|B)P(B) = P(B|A)P(A)$
    \begin{enumerate}
        \item [1.] $P(A|B) \geq 0$
        \item [2.] $P(B|B) = 1$
        \item [3.] If $A=\bigcup_{i=1}^{N}{A_i}$ with $A_i \cap A_j = \emptyset$ \\
        then $P(A|B) = \sum_{i=1}^{N} P(A_i|B)$
    \end{enumerate}
    \end{minipage}
};

        For any $A,B \subseteq S$:
        \begin{enumerate}
        \item[1.]$P(A^c) = 1 - P(A)$ 
        \item[2.]$P(A) \leq 1$ 
        \item[3.]$P(\emptyset) = 0$ 
        \item[4.]If $B = \bigcup_{i=1}^{n} B_i$ and $B_i \cap B_j = \emptyset$ and $i \neq j$, Then \\ $P(B) = \sum_{i=1}^{n} P(A_i)$ 
        \item[5.]$P(A\cup B) = P(A) + P(B) - P(A \cap B)$ 
        \item[6.]If $A \subseteq B$ then $P(A) \leq P(B)$
        \item[7.]For any event $A\subseteq B$ and event space $\{B_1, B_2, \ldots, B_n\}$ then $P(A) = \sum_{i=1}^{n} P(A \cap B_i)$
        \item[8.]If a sample space has outcomes that are equally likely, then for any event $A$, $P(A) = \frac{|A|}{|S|}$
        \end{enumerate}

%------------ Expected Value & Variance  ---------------
\begin{tikzpicture}
\node [mybox] (box){%
    \begin{minipage}{0.3\textwidth}
    $E[X]= \mu_X = \sum_{x\in S_X}^{}{}xP_X(x)$ = $\int_{-\infty}^{\infty}{xf_X(x)dx}$\\
    Derived random variable $Y = g(X)$:\\
    $E[Y]= \mu_Y = \sum_{x\in S_X}^{}{}g(x)P_X(x)$= $\int_{-\infty}^{\infty}{g(x)f_X(x)dx}$\\
    $Y = aX+b, a,b \in \mathbb{R}$ Then, $E[Y]=aE[X]+b$\\
    $Y = X- \mu_X$ Then, $E[Y]=E[X-\mu_X]=0$\\
    Variance:\\
    Var$(X)=\sum_{x\in S_X}^{}(x-\mu_X)^2P_X(x)$\\
    Var$(X)=\int_{-\infty}^{\infty}{(x-\mu_X)^2f_X(x)dx}$\\
    Var$(X)=\sigma_X^2=E[X^2]-(E[X])^2 \quad \sigma_X=\sqrt{\text{Var}(X)}$\\
    If Y = X+b, Var$(Y)$=Var$(X)\quad$ Var$(\text{Const})=0$\\
    If Y=aX, Var$(Y)$=$a^2$Var$(X)$
    \end{minipage}
};

%------------ Random Variables & PMF ---------------
\begin{tikzpicture}
\node [mybox] (box){%
    \begin{minipage}{0.3\textwidth}
    \small
    Discrete, finite and continuous(PDF)\\
    Random variables map Sample Space to $\mathbb{R}$: 
    
    $S = \{\zeta\}, S_X = {X(\zeta)\in \mathbb{R}|\zeta \in S}, S_X\subseteq \mathbb{R}$

    Equivalent Events: 
    $A = \{\zeta|X(\zeta)\in B\}$

    For a random variable $X$ w/ $P_X(x)$ and range $S_X$
    \begin{enumerate}
        \item[1.] For any x $P_X(x) \geq 0$
        \item[2.] $\sum_{x\in S_X}^{}{P_X(x)}=1$
        \item[3.] For any event $B \subset S_X, P(B) = \sum_{x\in S_B}^{}{P_X(x)}$ 
    \end{enumerate}
    \textbf{Common Discrete R.Vs:}
    \begin{center}
    \scriptsize
    \begin{tabular}{|l|l|l|}
    \hline
    \textbf{Name} & \textbf{PMF $P_X(x)$} & \textbf{E[x]} \\
    \hline
    Bernoulli &
    $\begin{cases}
    1-p, & x=0 \\
    p, & x=1 \\
    0, & \text{otherwise}
    \end{cases}$ &
    p \\

    \hline
    Geometric &
    $\begin{cases}
    p(1-p)^{x-1}, & x=1,2,\ldots \\
    0, & \text{otherwise}
    \end{cases}$ &
    $\frac{1}{p}$ \\

    \hline
    Binomial &
    $\begin{cases}
    \binom{n}{x}p^x(1-p)^{n-x}, & x=0,1,\ldots,n \\
    0, & \text{otherwise}
    \end{cases}$ &
    $np$ \\

    \hline
    Pascal &
    $\begin{cases}
    \binom{x-1}{k-1}p^k(1-p)^{x-k}, & x=k,k+1,\ldots \\
    0, & \text{otherwise}
    \end{cases}$ &
    $\frac{k}{p}$ \\

    \hline
    Uniform &
    $\begin{cases}
    \dfrac{1}{l-k+1}, & x=k,k+1,\ldots,l \\
    0, & \text{otherwise}
    \end{cases}$ &
    $\frac{k+l}{2}$ \\

    \hline
    Poisson &
    $\begin{cases}
    \dfrac{\alpha^x e^{-\alpha}}{x!}, & x=0,1,2,\ldots \\
    0, & \text{otherwise}
    \end{cases}$ &
    $\alpha$ \\
    \hline
    \end{tabular}
    \vspace{2pt}
    PDF: $\frac{dF_X(x)}{dx}=f_X(x)$
    \end{center}
    \end{minipage}
